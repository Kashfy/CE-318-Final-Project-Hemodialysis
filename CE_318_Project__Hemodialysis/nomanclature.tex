\subsection{Nomenclature}

\vspace{-0.3cm}
\begin{description}
    \setlength\itemsep{-0.15cm}
    \setlength\parskip{0pt}
    \setlength\parsep{0pt}

    \item[\(\dot{n}_A\)] Urea molar removal rate, \(\mathrm{mol/s}\).
    \item[\(J_A\)] Urea molar flux, \(\mathrm{mol/(m^2 \cdot s)}\).
    \item[\(Q_b, Q_d\)] Blood and dialysate flow rates, \(\mathrm{m^3/s}\).
    \item[\(C_A\)] Urea concentration, \(\mathrm{mol/m^3}\) or \(\mathrm{mg/L}\).
    \item[\(C_{b,i}, C_{b,o}\)] Blood urea concentration entering and leaving the dialyzer.
    \item[\(C_{d,i}, C_{d,o}\)] Dialysate urea concentration entering and leaving the dialyzer.

    \item[\(D_{AB}\)] Urea diffusivity in the liquid phase, \(\mathrm{m^2/s}\).
    \item[\(D_{\text{eff}}\)] Effective urea diffusivity through the membrane, \(\mathrm{m^2/s}\).
    \item[\(\varepsilon\)] Membrane void fraction.
    \item[\(\delta_m\)] Membrane thickness, \(\mathrm{m}\).
    \item[\(P_A\)] Urea membrane permeability.
    \item[\(S_A\)] Urea solubility coefficient in the membrane.

    \item[\(K_o\)] Overall mass-transfer coefficient, \(\mathrm{m/s}\).
    \item[\(k_b, k_d\)] Blood-side and dialysate-side mass-transfer coefficients, \(\mathrm{m/s}\).
    \item[\(A_m\)] Active membrane surface area, \(\mathrm{m^2}\).
    \item[\(\Delta C_{\text{lm}}\)] Log-mean concentration difference.
    \item[\(\Delta C_1, \Delta C_2\)] Concentration driving forces at each end of the dialyzer.

    \item[\(N_f\)] Number of hollow fibers.
    \item[\(d_f\)] Hollow fiber diameter, \(\mathrm{m}\).
    \item[\(L\)] Hollow fiber length, \(\mathrm{m}\).
    \item[\(u_b, u_d\)] Blood and dialysate velocities, \(\mathrm{m/s}\).

    \item[\(T, T_{\text{ref}}\)] Operating and reference temperatures, \(\mathrm{K}\).
    \item[\(\mu_T, \mu_{\text{ref}}\)] Operating and reference viscosities, \(\mathrm{Pa \cdot s}\).
    \item[\(f_{\text{free}}\)] Fraction of urea not protein-bound.
    \item[\(C_{A,\text{free}}\)] Free urea concentration available for diffusion.
    \item[\(Cl\)] Dialyzer clearance, \(\mathrm{m^3/s}\) or \(\mathrm{mL/min}\).
\end{description}
